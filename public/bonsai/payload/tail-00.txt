  function gameDaysBetween(from, to) { return Math.max(0, (to - from) / GAME_DAY_REAL_MS); }
  function gameDate(profileCreatedAt, at) {
    const days = gameDaysBetween(profileCreatedAt, at);
    const total = Math.floor(days);
    const year = Math.floor(total / 360) + 1;
    const dayOfYear = total % 360;
    const seasonIndex = Math.floor(dayOfYear / 90);
    const season = ['spring','summer','autumn','winter'][seasonIndex];
    const seasonJa = ['春','夏','秋','冬'][seasonIndex];
    const dayInSeason = dayOfYear % 90 + 1;
    return { days, total, year, dayOfYear, season, seasonJa, dayInSeason };
  }
  function weekStart(at) {
    const now = Number(at || Date.now());
    const jst = new Date(now + 9 * 3600000);
    const dayFromMonday = (jst.getUTCDay() + 6) % 7;
    const mondayJstMidnightUtc = Date.UTC(jst.getUTCFullYear(), jst.getUTCMonth(), jst.getUTCDate() - dayFromMonday, 0, 0, 0) - 9 * 3600000;
    return mondayJstMidnightUtc;
  }
